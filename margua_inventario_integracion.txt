function productOptions(){
  const active=productos.filter(p=>p.activo);
  $('productSelect').innerHTML='<option value="">Selecciona un producto del inventario</option>'+active.map(p=>`<option value="${p.id}">${escapeHtml(p.nombre)} · ${money(p.precio)} · Stock ${Number(p.stock_actual||0)}</option>`).join('');
  $('movementProduct').innerHTML='<option value="">Selecciona producto</option>'+active.map(p=>`<option value="${p.id}">${escapeHtml(p.nombre)} · Stock ${Number(p.stock_actual||0)}</option>`).join('');
}
function renderInventory(){
  if(!$('inventoryList')) return;
  productOptions();
  const q=($('inventorySearch')?.value||'').trim().toLowerCase();
  const filtered=productos.filter(p=>(p.nombre+' '+(p.categoria||'')).toLowerCase().includes(q));
  const list=$('inventoryList'); list.innerHTML='';
  if(!filtered.length){list.innerHTML='<div class="empty">No hay productos que coincidan con la búsqueda.</div>'}
  filtered.forEach(p=>{
    const low=Number(p.stock_actual||0)<=Number(p.stock_minimo||0);
    const d=document.createElement('div'); d.className='msg';
    d.innerHTML=`<div class="msghead"><b>${escapeHtml(p.nombre)}</b><span class="status ${low?'alert':'busy'}">${low?'Stock bajo':'Disponible'}</span></div><p>${escapeHtml(p.categoria||'Sin categoría')} · Venta: <b>${money(p.precio)}</b> · Costo: ${money(p.costo_compra)}<br>Existencias: <b>${Number(p.stock_actual||0)}</b> · Mínimo: ${Number(p.stock_minimo||0)}${p.activo?'':' · INACTIVO'}</p><div class="actions"><button class="btn secondary" data-edit="${p.id}">Editar</button><button class="btn ${p.activo?'secondary':'done'}" data-toggle="${p.id}">${p.activo?'Desactivar':'Activar'}</button></div>`;
    list.appendChild(d);
  });
  list.querySelectorAll('[data-edit]').forEach(b=>b.onclick=()=>editProduct(Number(b.dataset.edit)));
  list.querySelectorAll('[data-toggle]').forEach(b=>b.onclick=()=>toggleProduct(Number(b.dataset.toggle)));
  loadMovementHistory();
}
async function editProduct(id){
  const p=productos.find(x=>Number(x.id)===id); if(!p)return;
  const nombre=prompt('Nombre del producto:',p.nombre); if(nombre===null)return;
  if(!nombre.trim())return toast('El nombre no puede estar vacío');
  const categoria=prompt('Categoría:',p.categoria||''); if(categoria===null)return;
  const precio=prompt('Precio de venta (COP):',p.precio); if(precio===null)return;
  const costo=prompt('Costo de compra (COP):',p.costo_compra||0); if(costo===null)return;
  const minimo=prompt('Stock mínimo:',p.stock_minimo||0); if(minimo===null)return;
  if(!Number.isFinite(Number(precio))||Number(precio)<=0||!Number.isFinite(Number(costo))||Number(costo)<0||!Number.isFinite(Number(minimo))||Number(minimo)<0)return toast('Revisa los valores numéricos');
  const r=await db.from('productos').update({nombre:nombre.trim(),categoria:categoria.trim()||null,precio:Number(precio),costo_compra:Number(costo),stock_minimo:Number(minimo)}).eq('id',id);
  if(r.error)return toast('No se pudo editar: '+r.error.message);
  await loadData(); toast('Producto actualizado');
}
async function toggleProduct(id){
  const p=productos.find(x=>Number(x.id)===id); if(!p)return;
  const r=await db.from('productos').update({activo:!p.activo}).eq('id',id);
  if(r.error)return toast('No se pudo cambiar el estado: '+r.error.message);
  await loadData(); toast(p.activo?'Producto desactivado':'Producto activado');
}
async function loadMovementHistory(){
  const el=$('movementHistory'); if(!el)return;
  const r=await db.from('movimientos_inventario').select('id,producto_id,tipo,cantidad,stock_anterior,stock_nuevo,motivo,creado_at').order('creado_at',{ascending:false}).limit(30);
  if(r.error){el.innerHTML='<div class="empty">No se pudo leer el historial: '+escapeHtml(r.error.message)+'</div>';return}
  el.innerHTML='';
  if(!r.data?.length){el.innerHTML='<div class="empty">Todavía no hay movimientos.</div>';return}
  r.data.forEach(m=>{const p=productos.find(x=>Number(x.id)===Number(m.producto_id));const d=document.createElement('div');d.className='msg';d.innerHTML=`<div class="msghead"><b>${escapeHtml(p?.nombre||('Producto #'+m.producto_id))}</b><span class="tag">${escapeHtml(m.tipo.toUpperCase())}</span></div><p>Cantidad: ${Number(m.cantidad)} · Stock: ${Number(m.stock_anterior)} → ${Number(m.stock_nuevo)}<br>${escapeHtml(m.motivo||'Sin motivo')}</p><span class="muted">${new Date(m.creado_at).toLocaleString('es-CO')}</span>`;el.appendChild(d)});
}

NOTAS DE SEGURIDAD Y LIMITACIONES
- Las políticas RLS de productos deben permitir SELECT para authenticated y la política de gestión creada en el paso anterior.
- La tabla movimientos_inventario permite SELECT por política; las escrituras se hacen mediante la función RPC.
- El trigger de consumos se activa con INSERT y DELETE, no con UPDATE. Por ahora no edites cantidad ni producto de un consumo existente directamente.
- Si crear producto funciona pero falla el stock inicial, el producto queda creado con stock 0; vuelve a abrir Inventario y registra un ajuste con el stock correcto.
- No pongas la clave service_role en el HTML.
